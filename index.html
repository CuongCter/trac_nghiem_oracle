1 Giới thiệu dự án

🎯 Bối cảnh

Trong các cơ sở đào tạo hiện nay, việc tổ chức thi trắc nghiệm vẫn còn nhiều hạn chế: tốn thời gian in ấn đề, chấm bài thủ công; khó thống kê; khó đảm bảo tính công bằng và chống gian lận. Dự án ra đời nhằm số hóa toàn bộ quy trình Tạo đề → Phân công → Làm bài → Chấm điểm → Báo cáo.

2 Lược đồ cơ sở dữ liệu

📐 Sơ đồ ER (Entity Relationship)

Mô hình gồm 13 bảng chia thành 3 nhóm: danh mục (USERS · SUBJECTS · CLASSES), ngân hàng đề (QUESTIONS · EXAMS · EXAM_QUESTIONS · EXAM_CLASSES), và quá trình thi (EXAM_ATTEMPTS · ATTEMPT_ANSWERS · VIOLATIONS · EXAM_RESULTS · RESULT_ANSWERS).

3 Danh sách bảng & vai trò

Mỗi bảng đều có ID tự tăng (NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY), cùng các cột CREATED_AT / UPDATED_AT theo dõi lịch sử.

4 Ràng buộc toàn vẹn (Constraints)

Các ràng buộc CHECK giúp Oracle ngăn chặn dữ liệu rác ngay tầng database, không cần thêm validation phức tạp ở backend.

SQL · CHECK Constraints
-- Trên bảng USERS
ALTER TABLE USERS ADD CONSTRAINT USERS_ROLE_CHK   CHECK (ROLE   IN ('ADMIN','TEACHER','STUDENT'));
ALTER TABLE USERS ADD CONSTRAINT USERS_STATUS_CHK CHECK (STATUS IN ('ACTIVE','LOCKED'));

-- Trên bảng QUESTIONS
ALTER TABLE QUESTIONS ADD CONSTRAINT QUESTIONS_CORRECT_CHK CHECK (CORRECT_ANSWER IN ('A','B','C','D'));
ALTER TABLE QUESTIONS ADD CONSTRAINT QUESTIONS_DIFF_CHK    CHECK (DIFFICULTY     IN ('EASY','MEDIUM','HARD'));
ALTER TABLE QUESTIONS ADD CONSTRAINT QUESTIONS_POINT_CHK   CHECK (POINT > 0 AND POINT <= 100);

-- Trên bảng EXAMS
ALTER TABLE EXAMS ADD CONSTRAINT EXAMS_STATUS_CHK    CHECK (STATUS IN ('DRAFT','PUBLISHED','CLOSED'));
ALTER TABLE EXAMS ADD CONSTRAINT EXAMS_TIME_CHK      CHECK (END_TIME > START_TIME);
ALTER TABLE EXAMS ADD CONSTRAINT EXAMS_DURATION_CHK  CHECK (DURATION_MINUTES > 0);

-- Trên bảng EXAM_ATTEMPTS
ALTER TABLE EXAM_ATTEMPTS ADD CONSTRAINT ATTEMPTS_STATUS_CHK CHECK (STATUS IN ('IN_PROGRESS','SUBMITTED','EXPIRED'));

-- Trên bảng ATTEMPT_ANSWERS
ALTER TABLE ATTEMPT_ANSWERS ADD CONSTRAINT ATTEMPT_ANSWERS_OPT_CHK
  CHECK (SELECTED_OPTION IS NULL OR SELECTED_OPTION IN ('A','B','C','D'));
SQL · UNIQUE Constraints
-- Đảm bảo email không trùng
ALTER TABLE USERS           ADD CONSTRAINT USERS_EMAIL_UN    UNIQUE (EMAIL);
ALTER TABLE SUBJECTS        ADD CONSTRAINT SUBJECTS_CODE_UN  UNIQUE (CODE);
ALTER TABLE CLASSES         ADD CONSTRAINT CLASSES_NAME_UN   UNIQUE (NAME);

-- Mỗi sinh viên chỉ có một attempt và một kết quả cho mỗi đề
ALTER TABLE EXAM_ATTEMPTS   ADD CONSTRAINT ATTEMPTS_UN       UNIQUE (EXAM_ID, STUDENT_ID);
ALTER TABLE EXAM_RESULTS    ADD CONSTRAINT RESULTS_UN        UNIQUE (EXAM_ID, STUDENT_ID);

-- Khóa chính ghép cho bảng trung gian
ALTER TABLE CLASS_STUDENTS  ADD CONSTRAINT CLASS_STUDENTS_PK PRIMARY KEY (CLASS_ID, STUDENT_ID);
ALTER TABLE EXAM_QUESTIONS  ADD CONSTRAINT EXAM_QUESTIONS_PK PRIMARY KEY (EXAM_ID, QUESTION_ID);
ALTER TABLE EXAM_CLASSES    ADD CONSTRAINT EXAM_CLASSES_PK   PRIMARY KEY (EXAM_ID, CLASS_ID);
SQL · FOREIGN KEY (ràng buộc toàn vẹn tham chiếu)
-- Bảng SUBJECTS: người tạo có thể bị xóa → giữ môn học
ALTER TABLE SUBJECTS ADD CONSTRAINT SUBJECTS_CREATED_BY_FK
  FOREIGN KEY (CREATED_BY) REFERENCES USERS(ID) ON DELETE SET NULL;

-- Bảng QUESTIONS: nếu xóa môn học → xóa hết câu hỏi của môn đó
ALTER TABLE QUESTIONS ADD CONSTRAINT QUESTIONS_SUBJECT_FK
  FOREIGN KEY (SUBJECT_ID) REFERENCES SUBJECTS(ID) ON DELETE CASCADE;

-- Bảng EXAM_QUESTIONS, EXAM_ATTEMPTS ...: xóa đề → xóa hết dữ liệu liên quan
ALTER TABLE EXAM_QUESTIONS ADD CONSTRAINT EXAM_QUESTIONS_EXAM_FK
  FOREIGN KEY (EXAM_ID) REFERENCES EXAMS(ID) ON DELETE CASCADE;

ALTER TABLE EXAM_ATTEMPTS ADD CONSTRAINT ATTEMPTS_EXAM_FK
  FOREIGN KEY (EXAM_ID) REFERENCES EXAMS(ID) ON DELETE CASCADE;

ALTER TABLE EXAM_RESULTS ADD CONSTRAINT RESULTS_EXAM_FK
  FOREIGN KEY (EXAM_ID) REFERENCES EXAMS(ID) ON DELETE CASCADE;

-- Bảng CLASS_STUDENTS: cả hai phía đều dùng CASCADE
ALTER TABLE CLASS_STUDENTS ADD CONSTRAINT CLASS_STUDENTS_CLASS_FK
  FOREIGN KEY (CLASS_ID)   REFERENCES CLASSES(ID)  ON DELETE CASCADE;
ALTER TABLE CLASS_STUDENTS ADD CONSTRAINT CLASS_STUDENTS_USER_FK
  FOREIGN KEY (STUDENT_ID) REFERENCES USERS(ID)    ON DELETE CASCADE;

5 Chỉ mục (Indexes) — Tối ưu hiệu năng

Mỗi INDEX dưới đây phục vụ một kiểu truy vấn cụ thể trong hệ thống (lọc câu hỏi theo độ khó, lấy đề theo thời gian, ...).

SQL · CREATE INDEX
-- Tăng tốc lọc câu hỏi theo môn & độ khó (random đề thi)
CREATE INDEX IDX_QUESTIONS_SUBJECT     ON QUESTIONS(SUBJECT_ID);
CREATE INDEX IDX_QUESTIONS_DIFFICULTY  ON QUESTIONS(SUBJECT_ID, DIFFICULTY);

-- Tăng tốc lọc đề thi theo trạng thái + khung giờ
CREATE INDEX IDX_EXAMS_SUBJECT_STATUS  ON EXAMS(SUBJECT_ID, STATUS);
CREATE INDEX IDX_EXAMS_STATUS_START    ON EXAMS(STATUS, START_TIME, END_TIME);

-- Tăng tốc "sinh viên X đã làm đề nào", "đề Y do ai thi"
CREATE INDEX IDX_ATTEMPTS_STUDENT      ON EXAM_ATTEMPTS(STUDENT_ID);
CREATE INDEX IDX_ATTEMPTS_EXAM         ON EXAM_ATTEMPTS(EXAM_ID);
CREATE INDEX IDX_RESULTS_STUDENT       ON EXAM_RESULTS(STUDENT_ID);

-- Tăng tốc các truy vấn về lớp & thành viên
CREATE INDEX IDX_CLASS_STUDENTS_STUDENT ON CLASS_STUDENTS(STUDENT_ID);
CREATE INDEX IDX_EXAM_QUESTIONS_Q      ON EXAM_QUESTIONS(QUESTION_ID);
CREATE INDEX IDX_EXAM_CLASSES_CLASS    ON EXAM_CLASSES(CLASS_ID);

-- Truy vết vi phạm
CREATE INDEX IDX_VIOLATIONS_ATTEMPT    ON VIOLATIONS(ATTEMPT_ID, OCCURRED_AT);

-- Kiểm tra index đã được dùng bởi Oracle
EXPLAIN PLAN FOR
  SELECT * FROM EXAM_ATTEMPTS WHERE STUDENT_ID = 1 AND STATUS = 'IN_PROGRESS';

SELECT * FROM TABLE(DBMS_XPLAN.DISPLAY);

8 Các câu lệnh DDL

DDL (Data Definition Language) dùng để định nghĩa và quản lý cấu trúc cơ sở dữ liệu: CREATE, ALTER, DROP, TRUNCATE, RENAME, COMMENT, GRANT, ...

8.1. Tạo bảng (trích schema.sql)

SQL · CREATE TABLE
-- Tài khoản hệ thống (Admin / Teacher / Student)
CREATE TABLE USERS (
  ID            NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
  FULL_NAME     VARCHAR2(120 CHAR) NOT NULL,
  EMAIL         VARCHAR2(180 CHAR) NOT NULL,
  PASSWORD_HASH VARCHAR2(255 CHAR) NOT NULL,
  ROLE          VARCHAR2(20 CHAR)  NOT NULL,
  STATUS        VARCHAR2(20 CHAR)  DEFAULT 'ACTIVE' NOT NULL,
  CREATED_AT    TIMESTAMP          DEFAULT SYSTIMESTAMP NOT NULL,
  UPDATED_AT    TIMESTAMP          DEFAULT SYSTIMESTAMP NOT NULL,
  CONSTRAINT USERS_EMAIL_UN    UNIQUE (EMAIL),
  CONSTRAINT USERS_ROLE_CHK   CHECK (ROLE   IN ('ADMIN','TEACHER','STUDENT')),
  CONSTRAINT USERS_STATUS_CHK CHECK (STATUS IN ('ACTIVE','LOCKED'))
);

-- Ngân hàng câu hỏi trắc nghiệm A/B/C/D
CREATE TABLE QUESTIONS (
  ID             NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
  SUBJECT_ID     NUMBER NOT NULL,
  CONTENT        VARCHAR2(2000 CHAR) NOT NULL,
  OPTION_A       VARCHAR2(500 CHAR)  NOT NULL,
  OPTION_B       VARCHAR2(500 CHAR)  NOT NULL,
  OPTION_C       VARCHAR2(500 CHAR)  NOT NULL,
  OPTION_D       VARCHAR2(500 CHAR)  NOT NULL,
  CORRECT_ANSWER CHAR(1) NOT NULL,
  DIFFICULTY     VARCHAR2(20 CHAR)  NOT NULL,
  CHAPTER        VARCHAR2(80 CHAR),
  POINT          NUMBER(6,2)        DEFAULT 1 NOT NULL,
  CREATED_BY     NUMBER,
  CREATED_AT     TIMESTAMP          DEFAULT SYSTIMESTAMP NOT NULL,
  UPDATED_AT     TIMESTAMP          DEFAULT SYSTIMESTAMP NOT NULL,
  CONSTRAINT QUESTIONS_SUBJECT_FK    FOREIGN KEY (SUBJECT_ID) REFERENCES SUBJECTS(ID) ON DELETE CASCADE,
  CONSTRAINT QUESTIONS_CREATED_BY_FK FOREIGN KEY (CREATED_BY) REFERENCES USERS(ID)    ON DELETE SET NULL,
  CONSTRAINT QUESTIONS_CORRECT_CHK   CHECK (CORRECT_ANSWER IN ('A','B','C','D')),
  CONSTRAINT QUESTIONS_DIFF_CHK      CHECK (DIFFICULTY     IN ('EASY','MEDIUM','HARD')),
  CONSTRAINT QUESTIONS_POINT_CHK     CHECK (POINT > 0 AND POINT <= 100)
);

8.2. ALTER TABLE — Sửa bảng

SQL · ALTER TABLE
-- Thêm cột
ALTER TABLE EXAMS ADD PASS_SCORE NUMBER(4,2) DEFAULT 5 NOT NULL;

-- Thêm ràng buộc CHECK
ALTER TABLE USERS
  ADD CONSTRAINT USERS_NAME_LEN_CHK CHECK (LENGTH(FULL_NAME) >= 2);

-- Bật / tắt ràng buộc FK
ALTER TABLE QUESTIONS ENABLE  CONSTRAINT QUESTIONS_SUBJECT_FK;
ALTER TABLE QUESTIONS DISABLE CONSTRAINT QUESTIONS_SUBJECT_FK;

-- Đổi tên cột
ALTER TABLE EXAMS RENAME COLUMN DURATION_MINUTES TO DURATION_MIN;

-- Đổi tên bảng
ALTER TABLE CLASS_STUDENTS RENAME TO ENROLLMENTS;

8.3. Tạo VIEW

SQL · CREATE VIEW
-- View tổng hợp đề thi (dùng cho dashboard admin)
CREATE OR REPLACE VIEW V_EXAM_OVERVIEW AS
SELECT e.ID   AS EXAM_ID,
       e.TITLE,
       s.NAME AS SUBJECT_NAME,
       e.STATUS,
       e.START_TIME,
       e.END_TIME,
       (SELECT COUNT(*) FROM EXAM_QUESTIONS eq WHERE eq.EXAM_ID = e.ID) AS NUM_QUESTIONS,
       (SELECT COUNT(DISTINCT ec.CLASS_ID) FROM EXAM_CLASSES ec WHERE ec.EXAM_ID = e.ID) AS NUM_CLASSES,
       (SELECT COUNT(*) FROM EXAM_ATTEMPTS a WHERE a.EXAM_ID = e.ID) AS NUM_ATTEMPTS
FROM   EXAMS   e
JOIN   SUBJECTS s ON s.ID = e.SUBJECT_ID;

-- View bảng xếp hạng học viên
CREATE OR REPLACE VIEW V_STUDENT_RANKING AS
SELECT u.ID        AS STUDENT_ID,
       u.FULL_NAME,
       u.EMAIL,
       COUNT(r.ID) AS TOTAL_ATTEMPTS,
       ROUND(AVG(r.SCORE), 2) AS AVG_SCORE,
       SUM(CASE WHEN r.PASSED = 'Y' THEN 1 ELSE 0 END) AS PASSED_COUNT
FROM   USERS        u
LEFT   JOIN EXAM_RESULTS r ON r.STUDENT_ID = u.ID
WHERE  u.ROLE = 'STUDENT'
GROUP  BY u.ID, u.FULL_NAME, u.EMAIL;

9 Các câu lệnh DML

DML (Data Manipulation Language) gồm INSERT, UPDATE, DELETE, MERGE. Dưới đây là các thao tác thường gặp trong dự án.

9.1. INSERT — Thêm dữ liệu

SQL · INSERT
-- Thêm admin (mật khẩu đã được hash bằng bcrypt)
INSERT INTO USERS (FULL_NAME, EMAIL, PASSWORD_HASH, ROLE, STATUS)
VALUES ('Quản Trị Viên', 'admin@local',
        '$2a$10$abcdefghijklmnopqrstuv',  -- bcrypt hash của 'Admin@123'
        'ADMIN', 'ACTIVE');

-- Thêm môn học
INSERT INTO SUBJECTS (NAME, CODE, DESCRIPTION, CREATED_BY)
VALUES ('Lập trình Web', 'LTW', 'Môn học cơ sở', 1);

-- Thêm câu hỏi
INSERT INTO QUESTIONS (
  SUBJECT_ID, CONTENT, OPTION_A, OPTION_B, OPTION_C, OPTION_D,
  CORRECT_ANSWER, DIFFICULTY, CHAPTER, POINT, CREATED_BY
) VALUES (
  1,
  'Trong React, hook nào dùng để quản lý state?',
  'useEffect', 'useState', 'useRef', 'useMemo',
  'B', 'EASY', 'Hook cơ bản', 1, 1
);

-- Insert nhiều dòng cùng lúc bằng INSERT ALL
INSERT ALL
  INTO QUESTIONS (SUBJECT_ID, CONTENT, OPTION_A, OPTION_B, OPTION_C, OPTION_D,
                   CORRECT_ANSWER, DIFFICULTY, POINT, CREATED_BY)
  VALUES (1, 'Câu hỏi 1', 'A1', 'B1', 'C1', 'D1', 'A', 'EASY',   1, 1)
  INTO QUESTIONS (SUBJECT_ID, CONTENT, OPTION_A, OPTION_B, OPTION_C, OPTION_D,
                   CORRECT_ANSWER, DIFFICULTY, POINT, CREATED_BY)
  VALUES (1, 'Câu hỏi 2', 'A2', 'B2', 'C2', 'D2', 'B', 'MEDIUM', 2, 1)
SELECT 1 FROM DUAL;

COMMIT;

9.2. UPDATE — Cập nhật

SQL · UPDATE
-- Khóa tài khoản (soft lock thay vì xóa cứng)
UPDATE USERS
SET    STATUS = 'LOCKED', UPDATED_AT = SYSTIMESTAMP
WHERE  EMAIL = 'sinhvien_cantruycap@x.com';

-- Xuất bản đề thi: DRAFT → PUBLISHED
UPDATE EXAMS
SET    STATUS = 'PUBLISHED', UPDATED_AT = SYSTIMESTAMP
WHERE  ID = 10 AND STATUS = 'DRAFT';

-- Hết giờ → EXPIRED + auto-submit
UPDATE EXAM_ATTEMPTS
SET    STATUS = 'EXPIRED', SUBMITTED_AT = SYSTIMESTAMP
WHERE  STATUS = 'IN_PROGRESS'
  AND  END_TIME <= SYSTIMESTAMP;
COMMIT;

9.3. MERGE — Upsert

SQL · MERGE (UPSERT)
-- Lưu kết quả: nếu (EXAM_ID, STUDENT_ID) đã có → UPDATE, chưa có → INSERT
MERGE INTO EXAM_RESULTS tgt
USING (SELECT :exam_id AS EXAM_ID, :student_id AS STUDENT_ID FROM DUAL) src
ON    (tgt.EXAM_ID = src.EXAM_ID AND tgt.STUDENT_ID = src.STUDENT_ID)
WHEN MATCHED THEN
  UPDATE SET tgt.TOTAL_CORRECT = :tc,
             tgt.TOTAL_WRONG   = :tw,
             tgt.SCORE         = :score,
             tgt.PASSED        = :passed,
             tgt.GRADED_AT     = SYSTIMESTAMP
WHEN NOT MATCHED THEN
  INSERT (EXAM_ID, STUDENT_ID, TOTAL_CORRECT, TOTAL_WRONG, SCORE, PASSED)
  VALUES (:exam_id, :student_id, :tc, :tw, :score, :passed);
COMMIT;

10 Truy vấn dữ liệu (SELECT)

Tổng hợp các câu truy vấn quan trọng trong dự án: thống kê, báo cáo, ranking, phân tích gian lận, ...

10.1. Tổng quan hệ thống (dashboard admin)

SQL · Dashboard tổng quan
SELECT
  (SELECT COUNT(*) FROM USERS         WHERE ROLE = 'STUDENT')       AS TOTAL_STUDENTS,
  (SELECT COUNT(*) FROM USERS         WHERE ROLE = 'TEACHER')       AS TOTAL_TEACHERS,
  (SELECT COUNT(*) FROM SUBJECTS)                                    AS TOTAL_SUBJECTS,
  (SELECT COUNT(*) FROM QUESTIONS)                                   AS TOTAL_QUESTIONS,
  (SELECT COUNT(*) FROM EXAMS         WHERE STATUS = 'PUBLISHED')   AS PUBLISHED_EXAMS,
  (SELECT COUNT(*) FROM EXAM_ATTEMPTS)                                AS TOTAL_ATTEMPTS,
  (SELECT COUNT(*) FROM EXAM_RESULTS)                                 AS TOTAL_RESULTS,
  (SELECT ROUND(AVG(SCORE), 2) FROM EXAM_RESULTS)                    AS AVG_SCORE
FROM   DUAL;

10.2. Bảng xếp hạng học viên

SQL · Ranking
SELECT u.ID, u.FULL_NAME, u.EMAIL,
       COUNT(r.ID)             AS TOTAL_ATTEMPTS,
       ROUND(AVG(r.SCORE), 2)  AS AVG_SCORE,
       MAX(r.SCORE)            AS BEST_SCORE,
       SUM(CASE WHEN r.PASSED = 'Y' THEN 1 ELSE 0 END) AS PASS_COUNT
FROM   USERS         u
JOIN   EXAM_RESULTS  r ON r.STUDENT_ID = u.ID
JOIN   EXAMS         e ON e.ID         = r.EXAM_ID
JOIN   SUBJECTS      s ON s.ID         = e.SUBJECT_ID
WHERE  u.ROLE = 'STUDENT'
  AND  s.CODE = 'LTW'
GROUP  BY u.ID, u.FULL_NAME, u.EMAIL
HAVING COUNT(r.ID) > 0
ORDER  BY AVG_SCORE DESC NULLS LAST
FETCH  FIRST 10 ROWS ONLY;

11.4. Tự động chấm điểm khi SUBMIT (thay thế logic ở backend)

PL/SQL · Auto-grading trigger
CREATE OR REPLACE TRIGGER TRG_ATTEMPTS_AUTO_GRADE
AFTER UPDATE OF STATUS ON EXAM_ATTEMPTS
FOR EACH ROW
WHEN (NEW.STATUS IN ('SUBMITTED', 'EXPIRED'))
DECLARE
  v_total_correct         NUMBER := 0;
  v_total_wrong           NUMBER := 0;
  v_total_score           NUMBER := 0;
  v_result_id             NUMBER;
  c_max_score     CONSTANT NUMBER := 10;
BEGIN
  -- Đếm số câu đúng / sai
  FOR rec IN (
    SELECT aa.SELECTED_OPTION, q.CORRECT_ANSWER, q.POINT
    FROM   ATTEMPT_ANSWERS aa
    JOIN   QUESTIONS q ON q.ID = aa.QUESTION_ID
    WHERE  aa.ATTEMPT_ID = :NEW.ID
  ) LOOP
    IF rec.SELECTED_OPTION IS NOT NULL
       AND rec.SELECTED_OPTION = rec.CORRECT_ANSWER THEN
      v_total_correct := v_total_correct + 1;
      v_total_score   := v_total_score + rec.POINT;
    ELSIF rec.SELECTED_OPTION IS NOT NULL THEN
      v_total_wrong := v_total_wrong + 1;
    END IF;
  END LOOP;

  INSERT INTO EXAM_RESULTS (
    EXAM_ID, STUDENT_ID, TOTAL_CORRECT, TOTAL_WRONG,
    SCORE, PASSED, SUBMITTED_AT, GRADED_AT
  ) VALUES (
    :NEW.EXAM_ID, :NEW.STUDENT_ID, v_total_correct, v_total_wrong,
    v_total_score,
    CASE WHEN v_total_score >= (c_max_score / 2) THEN 'Y' ELSE 'N' END,
    SYSTIMESTAMP, SYSTIMESTAMP
  )
  RETURNING ID INTO v_result_id;

  INSERT INTO RESULT_ANSWERS (RESULT_ID, QUESTION_ID, SELECTED_OPTION, IS_CORRECT)
  SELECT v_result_id, aa.QUESTION_ID, aa.SELECTED_OPTION,
         CASE
           WHEN aa.SELECTED_OPTION IS NULL                  THEN 'N'
           WHEN aa.SELECTED_OPTION = q.CORRECT_ANSWER       THEN 'Y'
           ELSE 'N'
         END
  FROM   ATTEMPT_ANSWERS aa
  JOIN   QUESTIONS q ON q.ID = aa.QUESTION_ID
  WHERE  aa.ATTEMPT_ID = :NEW.ID;
END;
/

12 Procedure / Function

12.1. Function tính điểm

PL/SQL · FUNCTION
CREATE OR REPLACE FUNCTION FN_CALC_SCORE (p_attempt_id IN NUMBER)
  RETURN NUMBER
  DETERMINISTIC
IS
  v_total_earned   NUMBER := 0;
  v_total_possible NUMBER := 0;
  v_score10        NUMBER := 0;
  c_max_score CONSTANT NUMBER := 10;
BEGIN
  SELECT SUM(CASE
               WHEN aa.SELECTED_OPTION IS NOT NULL
                    AND aa.SELECTED_OPTION = q.CORRECT_ANSWER
               THEN q.POINT ELSE 0
             END),
         SUM(q.POINT)
  INTO   v_total_earned, v_total_possible
  FROM   ATTEMPT_ANSWERS aa
  JOIN   QUESTIONS q ON q.ID = aa.QUESTION_ID
  WHERE  aa.ATTEMPT_ID = p_attempt_id;

  IF v_total_possible > 0 THEN
    v_score10 := ROUND((v_total_earned / v_total_possible) * c_max_score, 2);
  END IF;

  RETURN v_score10;
END FN_CALC_SCORE;
/

-- Sử dụng
SELECT FN_CALC_SCORE(1) AS SCORE FROM DUAL;

12.2. Procedure chấm điểm batch

PL/SQL · PROCEDURE
CREATE OR REPLACE PROCEDURE SP_GRADE_ATTEMPT (
  p_attempt_id IN NUMBER,
  p_pass_score IN NUMBER DEFAULT 5
) AS
  v_score      NUMBER;
  v_correct    NUMBER := 0;
  v_wrong      NUMBER := 0;
  v_result_id  NUMBER;
BEGIN
  -- Đếm câu đúng / sai
  SELECT SUM(CASE
               WHEN aa.SELECTED_OPTION IS NOT NULL
                    AND aa.SELECTED_OPTION = q.CORRECT_ANSWER
               THEN 1 ELSE 0
             END),
         SUM(CASE
               WHEN aa.SELECTED_OPTION IS NOT NULL
                    AND aa.SELECTED_OPTION != q.CORRECT_ANSWER
               THEN 1 ELSE 0
             END)
  INTO   v_correct, v_wrong
  FROM   ATTEMPT_ANSWERS aa
  JOIN   QUESTIONS q ON q.ID = aa.QUESTION_ID
  WHERE  aa.ATTEMPT_ID = p_attempt_id;

  -- Tính điểm
  v_score := FN_CALC_SCORE(p_attempt_id);

  -- Lưu kết quả tổng quát
  INSERT INTO EXAM_RESULTS (EXAM_ID, STUDENT_ID, TOTAL_CORRECT, TOTAL_WRONG, SCORE, PASSED)
  SELECT a.EXAM_ID, a.STUDENT_ID, v_correct, v_wrong, v_score,
         CASE WHEN v_score >= p_pass_score THEN 'Y' ELSE 'N' END
  FROM   EXAM_ATTEMPTS a WHERE a.ID = p_attempt_id
  RETURNING ID INTO v_result_id;

  -- Lưu chi tiết từng đáp án
  INSERT INTO RESULT_ANSWERS (RESULT_ID, QUESTION_ID, SELECTED_OPTION, IS_CORRECT)
  SELECT v_result_id, aa.QUESTION_ID, aa.SELECTED_OPTION,
         CASE
           WHEN aa.SELECTED_OPTION IS NULL            THEN 'N'
           WHEN aa.SELECTED_OPTION = q.CORRECT_ANSWER THEN 'Y'
           ELSE 'N'
         END
  FROM   ATTEMPT_ANSWERS aa
  JOIN   QUESTIONS q ON q.ID = aa.QUESTION_ID
  WHERE  aa.ATTEMPT_ID = p_attempt_id;

  COMMIT;
END SP_GRADE_ATTEMPT;
/

-- Gọi thử

12.3. Procedure random câu hỏi theo độ khó

PL/SQL · Random + REF CURSOR
CREATE OR REPLACE PROCEDURE SP_RANDOM_QUESTIONS (
  p_subject_id   IN  NUMBER,
  p_easy_count   IN  NUMBER,
  p_medium_count IN  NUMBER,
  p_hard_count   IN  NUMBER,
  p_result       OUT SYS_REFCURSOR
) AS
BEGIN
  OPEN p_result FOR
    SELECT * FROM (
      SELECT q.*, 'EASY' AS DIFF_TAG,
             ROW_NUMBER() OVER (ORDER BY DBMS_RANDOM.VALUE) rn
      FROM QUESTIONS q
      WHERE SUBJECT_ID = p_subject_id AND DIFFICULTY = 'EASY'
    ) WHERE rn <= p_easy_count
    UNION ALL
    SELECT * FROM (
      SELECT q.*, 'MEDIUM' AS DIFF_TAG,
             ROW_NUMBER() OVER (ORDER BY DBMS_RANDOM.VALUE) rn
      FROM QUESTIONS q
      WHERE SUBJECT_ID = p_subject_id AND DIFFICULTY = 'MEDIUM'
    ) WHERE rn <= p_medium_count
    UNION ALL
    SELECT * FROM (
      SELECT q.*, 'HARD' AS DIFF_TAG,
             ROW_NUMBER() OVER (ORDER BY DBMS_RANDOM.VALUE) rn
      FROM QUESTIONS q
      WHERE SUBJECT_ID = p_subject_id AND DIFFICULTY = 'HARD'
    ) WHERE rn <= p_hard_count;
END SP_RANDOM_QUESTIONS;
/

12.4. Procedure đóng đề hết hạn (kết hợp DBMS_SCHEDULER)

PL/SQL · Job định kỳ
CREATE OR REPLACE PROCEDURE SP_CLOSE_PAST_EXAMS AS
BEGIN
  -- Đóng đề đã hết giờ
  UPDATE EXAMS
  SET    STATUS = 'CLOSED', UPDATED_AT = SYSTIMESTAMP
  WHERE  STATUS = 'PUBLISHED'
    AND  END_TIME < SYSTIMESTAMP;

  -- Auto-submit các attempt quá hạn
  UPDATE EXAM_ATTEMPTS
  SET    STATUS = 'EXPIRED', SUBMITTED_AT = SYSTIMESTAMP
  WHERE  STATUS = 'IN_PROGRESS'
    AND  END_TIME <= SYSTIMESTAMP;

  COMMIT;
END SP_CLOSE_PAST_EXAMS;
/

-- Tạo job chạy mỗi 5 phút
BEGIN
  DBMS_SCHEDULER.CREATE_JOB (
    job_name        => 'JOB_CLOSE_PAST_EXAMS',
    job_type        => 'PLSQL_BLOCK',
    job_action      => 'BEGIN SP_CLOSE_PAST_EXAMS; END;',
    repeat_interval => 'FREQ=MINUTELY; INTERVAL=5',
    enabled         => TRUE,
    comments        => 'Đóng các đề và attempt đã quá hạn'
  );
END;
/

Identity ColumnKhóa chính tự tăng cho 13 bảng, tương đương AUTO_INCREMENT.
